<body>

    <script>
        let valor;
        valor = parseInt(prompt('Ingrese un valor comprendido entre 1 y 5:'));
        switch (valor) {
            case 1:
                document.write('uno');
                break;
            case 2:
                document.write('dos');
                break;
            case 3:
                document.write('tres');
                break;
            case 4:
                document.write('cuatro');
                break;
            case 5:
                document.write('cinco');
                break;
            default:
                document.write('debe ingresar un valor comprendido entre 1 y 5.');
        }
    </script>

</body>

//11 DO WHILE
<!DOCTYPE html>
<html>

<head>
    <title>Ejemplo de JavaScript</title>
    <meta charset="UTF-8">
</head>

<body>

    <script>
        let valor;
        do {
            valor = parseInt(prompt('Ingrese un valor entre 0 y 999:', ''));
            document.write('El valor ' + valor + ' tiene ');
            if (valor < 10) {
                document.write('Tiene 1 digitos');
            } else {
                if (valor < 100) {
                    document.write('Tiene 2 digitos');
                } else {
                    document.write('Tiene 3 digitos');
                }
            }
            document.write('<br>');
        } while (valor != 0);
    </script>

</body>

</html>
